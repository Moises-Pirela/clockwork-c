cmake_minimum_required(VERSION 3.16)
project(clockwork VERSION 1.1.0 LANGUAGES C)

if(MSVC AND NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
    set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()

option(CW_BUILD_TESTS "Build Clockwork regression tests" ON)
option(CW_BUILD_EXAMPLES "Build the examples that are present in the tree" ON)
option(CW_BUILD_GAMES "Build the game modules under games/ as shared libraries" ON)
option(CW_BUILD_GUI "Build the Sokol tools: the editor, the player and render2d (bundled Sokol)" ON)
option(CW_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

# ---------------------------------------------------------------------------
# The engine: ECS core, reflection, scenes, game modules
# ---------------------------------------------------------------------------
add_library(clockwork STATIC
    src/clockwork.c src/cw_sim.c src/cw_commands.c src/cw_bytes.c src/cw_snapshot.c src/cw_file.c
    src/cw_reflect.c src/cw_scene.c src/cw_game.c src/cw_assets.c src/cw_anim.c src/cw_ui.c src/cw_brush.c)
add_library(clockwork::clockwork ALIAS clockwork)
target_include_directories(clockwork PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
set_target_properties(clockwork PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO POSITION_INDEPENDENT_CODE ON)
target_compile_features(clockwork PUBLIC c_std_11)
if(MSVC)
    target_compile_options(clockwork PRIVATE /W4 /WX)
else()
    target_compile_options(clockwork PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(clockwork PUBLIC m ${CMAKE_DL_LIBS})
endif()
if(CW_SANITIZE)
    if(MSVC)
        target_compile_options(clockwork PUBLIC /fsanitize=address)
    else()
        target_compile_options(clockwork PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(clockwork PUBLIC -fsanitize=address,undefined)
    endif()
endif()
function(cw_strict target)
    set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endfunction()

# ---------------------------------------------------------------------------
# Game modules: each project under games/ becomes a shared library at the
# path its .cwproj names (games/<name>/build/<name>_game.<suffix>), which is
# what the editor and the player load. tools/build-game.sh does the same
# from inside the editor.
# ---------------------------------------------------------------------------
set(CW_COMBAT_PROJECT "${CMAKE_CURRENT_SOURCE_DIR}/games/combat/combat.cwproj")
set(CW_ORBIT_PROJECT "${CMAKE_CURRENT_SOURCE_DIR}/games/orbit/orbit.cwproj")
set(CW_SANDBOX3D_PROJECT "${CMAKE_CURRENT_SOURCE_DIR}/games/sandbox3d/sandbox3d.cwproj")
function(cw_game_module name)
    add_library(${name}_game MODULE games/${name}/${name}_game.c)
    target_link_libraries(${name}_game PRIVATE clockwork)
    cw_strict(${name}_game)
    set(out "${CMAKE_CURRENT_SOURCE_DIR}/games/${name}/build")
    set_target_properties(${name}_game PROPERTIES PREFIX "" OUTPUT_NAME ${name}_game
        LIBRARY_OUTPUT_DIRECTORY "${out}" LIBRARY_OUTPUT_DIRECTORY_DEBUG "${out}" LIBRARY_OUTPUT_DIRECTORY_RELEASE "${out}")
    # the output directory lives in the project, outside the build tree: create it before linking
    add_custom_command(TARGET ${name}_game PRE_LINK COMMAND ${CMAKE_COMMAND} -E make_directory "${out}")
    if(APPLE)
        set_target_properties(${name}_game PROPERTIES SUFFIX ".dylib")
    endif()
endfunction()
if(CW_BUILD_GAMES)
    cw_game_module(combat)
    cw_game_module(orbit)
    cw_game_module(sandbox3d)
endif()

# Desktop tools: render2d, the player and the editor

if(CW_BUILD_GUI)
    set(CW_GUI_OK TRUE)
    if(UNIX AND NOT APPLE)
        # The X11 and OpenGL headers are bundled (third_party/x11, third_party/gl),
        # so no development package is needed: only the runtime libraries,
        # linked by file name when the development symlinks are absent.
        find_library(CW_X11_LIBRARY NAMES X11 libX11.so.6)
        find_library(CW_XI_LIBRARY NAMES Xi libXi.so.6)
        find_library(CW_XCURSOR_LIBRARY NAMES Xcursor libXcursor.so.1)
        find_library(CW_GL_LIBRARY NAMES GL libGL.so.1)
        if(NOT CW_X11_LIBRARY OR NOT CW_XI_LIBRARY OR NOT CW_XCURSOR_LIBRARY OR NOT CW_GL_LIBRARY)
            message(WARNING "CW_BUILD_GUI: a runtime library is missing (need libX11.so.6, libXi.so.6, libXcursor.so.1 and libGL.so.1); skipping the Sokol tools")
            set(CW_GUI_OK FALSE)
        endif()
    endif()
    if(CW_GUI_OK)
        include("${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare-sokol.cmake")
        add_library(clockwork_sokol STATIC examples/sokol_impl.c)
        target_compile_features(clockwork_sokol PUBLIC c_std_11)
        target_include_directories(clockwork_sokol SYSTEM PUBLIC "${CW_SOKOL}")
        if(MSVC)
            target_compile_options(clockwork_sokol PRIVATE /W0)
        else()
            target_compile_options(clockwork_sokol PRIVATE -w)
        endif()
        if(WIN32)
            target_compile_definitions(clockwork_sokol PUBLIC SOKOL_D3D11)
            target_link_libraries(clockwork_sokol PUBLIC d3d11 dxgi user32 gdi32 shell32 ole32)
        elseif(APPLE)
            enable_language(OBJC)
            set_source_files_properties(examples/sokol_impl.c PROPERTIES LANGUAGE OBJC)
            target_compile_definitions(clockwork_sokol PUBLIC SOKOL_METAL)
            target_link_libraries(clockwork_sokol PUBLIC "-framework Cocoa" "-framework QuartzCore" "-framework Metal" "-framework MetalKit")
        elseif(UNIX)
            foreach(CW_BUNDLED_HEADER IN ITEMS third_party/gl/GL/gl.h third_party/x11/X11/Xlib.h third_party/x11/X11/extensions/XInput2.h third_party/x11/X11/Xcursor/Xcursor.h)
                if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${CW_BUNDLED_HEADER}")
                    message(FATAL_ERROR "Bundled header missing: ${CW_BUNDLED_HEADER}. Restore third_party/gl and third_party/x11 from the repository.")
                endif()
            endforeach()
            target_compile_definitions(clockwork_sokol PUBLIC SOKOL_GLCORE)
            target_include_directories(clockwork_sokol BEFORE PRIVATE
                "${CMAKE_CURRENT_SOURCE_DIR}/third_party/x11" "${CMAKE_CURRENT_SOURCE_DIR}/third_party/gl")
            target_link_libraries(clockwork_sokol PUBLIC ${CW_X11_LIBRARY} ${CW_XCURSOR_LIBRARY} ${CW_XI_LIBRARY} ${CW_GL_LIBRARY} m pthread ${CMAKE_DL_LIBS})
        endif()

        add_library(clockwork_render2d STATIC render2d/render2d.c render2d/r2d_particles.c render2d/r2d_effects.c
            render2d/r2d_tunables.c render2d/r2d_post_params.c render2d/r2d_post.c render2d/r2d_post_shaders.c
            render2d/r2d_shadow.c render2d/r2d_lighting.c render2d/r2d_quality.c
            render2d/r2d_image.c render2d/r2d_material.c render2d/r2d_sprite.c render2d/r2d_sprite_shaders.c)
        add_library(clockwork::render2d ALIAS clockwork_render2d)
        target_include_directories(clockwork_render2d PUBLIC
            $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/render2d>
            $<INSTALL_INTERFACE:include>)
        target_link_libraries(clockwork_render2d PUBLIC clockwork_sokol)
        cw_strict(clockwork_render2d)

        # render3d: the unlit 3D renderer (primitives, camera, picking), composited into render2d's frame
        add_library(clockwork_render3d STATIC render3d/render3d.c render3d/r3d_mesh.c render3d/r3d_shaders.c)
        add_library(clockwork::render3d ALIAS clockwork_render3d)
        target_include_directories(clockwork_render3d PUBLIC
            $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/render3d>
            $<INSTALL_INTERFACE:include>)
        target_link_libraries(clockwork_render3d PUBLIC clockwork_sokol)
        cw_strict(clockwork_render3d)

        # render3d on its own: the four primitives in a window (a usage example and a smoke test)
        add_executable(clockwork_shapes3d examples/shapes3d.c)
        target_link_libraries(clockwork_shapes3d PRIVATE clockwork_render3d)
        cw_strict(clockwork_shapes3d)

        # the host side shared by the tools: the module canvas over render2d (and render3d for 3D views) and small widgets
        add_library(clockwork_host STATIC runtime/host_canvas.c runtime/host_view3d.c runtime/host_ui.c runtime/host_dock.c)
        target_include_directories(clockwork_host PUBLIC runtime)
        target_link_libraries(clockwork_host PUBLIC clockwork clockwork_render2d clockwork_render3d)
        cw_strict(clockwork_host)

        add_executable(clockwork_player runtime/player.c)
        target_link_libraries(clockwork_player PRIVATE clockwork_host)
        cw_strict(clockwork_player)

        add_executable(clockwork_editor editor/editor_main.c editor/editor_core.c editor/editor_settings.c editor/editor_anim.c)
        target_include_directories(clockwork_editor PRIVATE editor)
        target_compile_definitions(clockwork_editor PRIVATE CW_ENGINE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
        target_link_libraries(clockwork_editor PRIVATE clockwork_host)
        cw_strict(clockwork_editor)
        add_executable(clockwork_launcher launcher/launcher_main.c launcher/launcher_core.c)
        target_include_directories(clockwork_launcher PRIVATE launcher)
        target_compile_definitions(clockwork_launcher PRIVATE CW_ENGINE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
        target_link_libraries(clockwork_launcher PRIVATE clockwork_host)
        cw_strict(clockwork_launcher)
        # the product's front door: the same launcher under its own name (a windowed application on Windows)
        add_executable(Clockwork launcher/launcher_main.c launcher/launcher_core.c)
        target_include_directories(Clockwork PRIVATE launcher)
        target_compile_definitions(Clockwork PRIVATE CW_ENGINE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
        target_link_libraries(Clockwork PRIVATE clockwork_host)
        cw_strict(Clockwork)
        if(WIN32)
            set_target_properties(Clockwork PROPERTIES WIN32_EXECUTABLE ON LINK_FLAGS "/ENTRY:mainCRTStartup")
        endif()
        # the post-process tunables live next to the tools
        add_custom_command(TARGET clockwork_editor POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/tunable.cfg" "$<TARGET_FILE_DIR:clockwork_editor>/tunable.cfg")

        if(CW_BUILD_EXAMPLES AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/examples/save_load.c")
            add_executable(clockwork_save_load examples/save_load.c)
            target_link_libraries(clockwork_save_load PRIVATE clockwork clockwork_sokol)
        endif()
    endif()
endif()
