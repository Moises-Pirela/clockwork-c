cmake_minimum_required(VERSION 3.16)
project(clockwork VERSION 1.0.0 LANGUAGES C)

option(CW_BUILD_TESTS "Build Clockwork regression tests" ON)
option(CW_BUILD_EXAMPLES "Build the movement example" ON)
option(CW_BUILD_GUI "Build the interactive save/load tool (downloads pinned Sokol)" ON)
option(CW_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

add_library(clockwork STATIC src/clockwork.c src/cw_sim.c src/cw_commands.c src/cw_bytes.c src/cw_snapshot.c src/cw_file.c)
add_library(clockwork::clockwork ALIAS clockwork)
target_include_directories(clockwork PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
set_target_properties(clockwork PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_compile_features(clockwork PUBLIC c_std_11)
if(MSVC)
    target_compile_options(clockwork PRIVATE /W4 /WX)
else()
    target_compile_options(clockwork PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(CW_SANITIZE)
    if(MSVC)
        target_compile_options(clockwork PUBLIC /fsanitize=address)
    else()
        target_compile_options(clockwork PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(clockwork PUBLIC -fsanitize=address,undefined)
    endif()
endif()

if(CW_BUILD_EXAMPLES)
    add_executable(clockwork_movement examples/movement.c)
    target_link_libraries(clockwork_movement PRIVATE clockwork)
    if(CW_BUILD_GUI)
        set(CW_GUI_OK TRUE)
        if(UNIX AND NOT APPLE)
            find_package(X11)
            find_package(OpenGL)
            if(NOT X11_FOUND OR NOT X11_Xcursor_INCLUDE_PATH OR NOT X11_Xi_INCLUDE_PATH OR NOT OPENGL_FOUND)
                message(WARNING "CW_BUILD_GUI: X11, Xcursor, Xi or OpenGL development files missing; skipping clockwork_save_load")
                set(CW_GUI_OK FALSE)
            endif()
        endif()
        if(CW_GUI_OK)
            include("${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare-sokol.cmake")
            add_library(clockwork_sokol STATIC examples/sokol_impl.c)
            target_compile_features(clockwork_sokol PUBLIC c_std_11)
            target_include_directories(clockwork_sokol SYSTEM PUBLIC "${CW_SOKOL}")
            if(MSVC)
                target_compile_options(clockwork_sokol PRIVATE /W0)
            else()
                target_compile_options(clockwork_sokol PRIVATE -w)
            endif()
            if(WIN32)
                target_compile_definitions(clockwork_sokol PUBLIC SOKOL_D3D11)
                target_link_libraries(clockwork_sokol PUBLIC d3d11 dxgi user32 gdi32 shell32 ole32)
            elseif(APPLE)
                enable_language(OBJC)
                set_source_files_properties(examples/sokol_impl.c PROPERTIES LANGUAGE OBJC)
                target_compile_definitions(clockwork_sokol PUBLIC SOKOL_METAL)
                target_link_libraries(clockwork_sokol PUBLIC "-framework Cocoa" "-framework QuartzCore" "-framework Metal" "-framework MetalKit")
            elseif(UNIX)
                target_compile_definitions(clockwork_sokol PUBLIC SOKOL_GLCORE)
                target_include_directories(clockwork_sokol PRIVATE ${X11_INCLUDE_DIR} ${X11_Xcursor_INCLUDE_PATH} ${X11_Xi_INCLUDE_PATH})
                target_link_libraries(clockwork_sokol PUBLIC ${X11_LIBRARIES} ${X11_Xcursor_LIB} ${X11_Xi_LIB} OpenGL::GL m pthread ${CMAKE_DL_LIBS})
            endif()
            add_executable(clockwork_save_load examples/save_load.c)
            target_link_libraries(clockwork_save_load PRIVATE clockwork clockwork_sokol)
        endif()
    endif()
    if(UNIX)
        add_executable(clockwork_visualizer examples/visualizer.c)
        target_link_libraries(clockwork_visualizer PRIVATE clockwork m)
    endif()
endif()
if(CW_BUILD_TESTS)
    enable_testing()
    add_executable(clockwork_tests tests/test_clockwork.c)
    target_link_libraries(clockwork_tests PRIVATE clockwork)
    add_test(NAME clockwork_tests COMMAND clockwork_tests)
    if(CW_BUILD_EXAMPLES)
        add_test(NAME clockwork_movement COMMAND clockwork_movement)
        set_tests_properties(clockwork_movement PROPERTIES PASS_REGULAR_EXPRESSION "After 30 ticks: \\(3.00, 1.00\\)")
        if(TARGET clockwork_save_load)
            add_test(NAME clockwork_save_load COMMAND clockwork_save_load --headless ${CMAKE_CURRENT_BINARY_DIR}/clockwork_save_load_test.cwsn)
            set_tests_properties(clockwork_save_load PROPERTIES PASS_REGULAR_EXPRESSION "OK: save/load verified")
        endif()
        if(UNIX)
            add_test(NAME clockwork_visualizer COMMAND clockwork_visualizer --frames 300)
            set_tests_properties(clockwork_visualizer PROPERTIES PASS_REGULAR_EXPRESSION "OK: [0-9]+ checks, 0 violations")
        endif()
    endif()
endif()

include(GNUInstallDirs)
install(TARGETS clockwork EXPORT clockworkTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/clockwork.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT clockworkTargets NAMESPACE clockwork:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/clockwork)
