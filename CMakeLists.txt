cmake_minimum_required(VERSION 3.16)
project(clockwork VERSION 1.0.0 LANGUAGES C)

option(CW_BUILD_TESTS "Build Clockwork regression tests" ON)
option(CW_BUILD_EXAMPLES "Build the movement example" ON)
option(CW_BUILD_GUI "Build the Sokol tools: save/load, render2d and the combat demo (bundled Sokol)" ON)
option(CW_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

add_library(clockwork STATIC src/clockwork.c src/cw_sim.c src/cw_commands.c src/cw_bytes.c src/cw_snapshot.c src/cw_file.c)
add_library(clockwork::clockwork ALIAS clockwork)
target_include_directories(clockwork PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
set_target_properties(clockwork PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_compile_features(clockwork PUBLIC c_std_11)
if(MSVC)
    target_compile_options(clockwork PRIVATE /W4 /WX)
else()
    target_compile_options(clockwork PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(CW_SANITIZE)
    if(MSVC)
        target_compile_options(clockwork PUBLIC /fsanitize=address)
    else()
        target_compile_options(clockwork PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(clockwork PUBLIC -fsanitize=address,undefined)
    endif()
endif()

if(CW_BUILD_EXAMPLES)
    add_executable(clockwork_movement examples/movement.c)
    target_link_libraries(clockwork_movement PRIVATE clockwork)
    if(CW_BUILD_GUI)
        set(CW_GUI_OK TRUE)
        if(UNIX AND NOT APPLE)
            # The X11 and OpenGL headers are bundled (third_party/x11, third_party/gl),
            # so no development package is needed: only the runtime libraries,
            # linked by file name when the development symlinks are absent.
            find_library(CW_X11_LIBRARY NAMES X11 libX11.so.6)
            find_library(CW_XI_LIBRARY NAMES Xi libXi.so.6)
            find_library(CW_XCURSOR_LIBRARY NAMES Xcursor libXcursor.so.1)
            find_library(CW_GL_LIBRARY NAMES GL libGL.so.1)
            if(NOT CW_X11_LIBRARY OR NOT CW_XI_LIBRARY OR NOT CW_XCURSOR_LIBRARY OR NOT CW_GL_LIBRARY)
                message(WARNING "CW_BUILD_GUI: a runtime library is missing (need libX11.so.6, libXi.so.6, libXcursor.so.1 and libGL.so.1); skipping the Sokol tools")
                set(CW_GUI_OK FALSE)
            endif()
        endif()
        if(CW_GUI_OK)
            include("${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare-sokol.cmake")
            add_library(clockwork_sokol STATIC examples/sokol_impl.c)
            target_compile_features(clockwork_sokol PUBLIC c_std_11)
            target_include_directories(clockwork_sokol SYSTEM PUBLIC "${CW_SOKOL}")
            if(MSVC)
                target_compile_options(clockwork_sokol PRIVATE /W0)
            else()
                target_compile_options(clockwork_sokol PRIVATE -w)
            endif()
            if(WIN32)
                target_compile_definitions(clockwork_sokol PUBLIC SOKOL_D3D11)
                target_link_libraries(clockwork_sokol PUBLIC d3d11 dxgi user32 gdi32 shell32 ole32)
            elseif(APPLE)
                enable_language(OBJC)
                set_source_files_properties(examples/sokol_impl.c PROPERTIES LANGUAGE OBJC)
                target_compile_definitions(clockwork_sokol PUBLIC SOKOL_METAL)
                target_link_libraries(clockwork_sokol PUBLIC "-framework Cocoa" "-framework QuartzCore" "-framework Metal" "-framework MetalKit")
            elseif(UNIX)
                foreach(CW_BUNDLED_HEADER IN ITEMS third_party/gl/GL/gl.h third_party/x11/X11/Xlib.h third_party/x11/X11/extensions/XInput2.h third_party/x11/X11/Xcursor/Xcursor.h)
                    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${CW_BUNDLED_HEADER}")
                        message(FATAL_ERROR "Bundled header missing: ${CW_BUNDLED_HEADER}. Restore third_party/gl and third_party/x11 from the repository.")
                    endif()
                endforeach()
                target_compile_definitions(clockwork_sokol PUBLIC SOKOL_GLCORE)
                target_include_directories(clockwork_sokol BEFORE PRIVATE
                    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/x11" "${CMAKE_CURRENT_SOURCE_DIR}/third_party/gl")
                target_link_libraries(clockwork_sokol PUBLIC ${CW_X11_LIBRARY} ${CW_XCURSOR_LIBRARY} ${CW_XI_LIBRARY} ${CW_GL_LIBRARY} m pthread ${CMAKE_DL_LIBS})
            endif()
            add_executable(clockwork_save_load examples/save_load.c)
            target_link_libraries(clockwork_save_load PRIVATE clockwork clockwork_sokol)

            add_library(clockwork_render2d STATIC render2d/render2d.c render2d/r2d_particles.c render2d/r2d_effects.c
                render2d/r2d_tunables.c render2d/r2d_post_params.c render2d/r2d_post.c render2d/r2d_post_shaders.c
                render2d/r2d_shadow.c render2d/r2d_lighting.c render2d/r2d_quality.c)
            add_library(clockwork::render2d ALIAS clockwork_render2d)
            target_include_directories(clockwork_render2d PUBLIC
                $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/render2d>
                $<INSTALL_INTERFACE:include>)
            target_link_libraries(clockwork_render2d PUBLIC clockwork_sokol)
            set_target_properties(clockwork_render2d PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
            if(MSVC)
                target_compile_options(clockwork_render2d PRIVATE /W4 /WX)
            else()
                target_compile_options(clockwork_render2d PRIVATE -Wall -Wextra -Wpedantic -Werror)
            endif()
            add_executable(clockwork_combat examples/combat.c)
            target_link_libraries(clockwork_combat PRIVATE clockwork clockwork_render2d)
            # the post-process tunables live next to the demo binary
            add_custom_command(TARGET clockwork_combat POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/tunable.cfg" "$<TARGET_FILE_DIR:clockwork_combat>/tunable.cfg")
        endif()
    endif()
    if(UNIX)
        add_executable(clockwork_visualizer examples/visualizer.c)
        target_link_libraries(clockwork_visualizer PRIVATE clockwork m)
    endif()
endif()
if(CW_BUILD_TESTS)
    enable_testing()
    add_executable(clockwork_tests tests/test_clockwork.c)
    target_link_libraries(clockwork_tests PRIVATE clockwork)
    add_test(NAME clockwork_tests COMMAND clockwork_tests)
    # The render2d math tests only need the headers, not a graphics backend.
    include("${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare-sokol.cmake")
    add_executable(clockwork_render2d_tests tests/test_render2d.c)
    target_include_directories(clockwork_render2d_tests PRIVATE render2d)
    target_include_directories(clockwork_render2d_tests SYSTEM PRIVATE "${CW_SOKOL}")
    target_compile_features(clockwork_render2d_tests PRIVATE c_std_11)
    if(NOT MSVC)
        target_link_libraries(clockwork_render2d_tests PRIVATE m)
    endif()
    add_test(NAME clockwork_render2d_tests COMMAND clockwork_render2d_tests)
    set_tests_properties(clockwork_render2d_tests PROPERTIES PASS_REGULAR_EXPRESSION "OK: [0-9]+ render2d checks passed")
    # Particle and effect systems: their simulation is renderer-free; the test stubs the draw calls.
    add_executable(clockwork_r2d_fx_tests tests/test_r2d_fx.c render2d/r2d_particles.c render2d/r2d_effects.c)
    target_include_directories(clockwork_r2d_fx_tests PRIVATE render2d)
    target_include_directories(clockwork_r2d_fx_tests SYSTEM PRIVATE "${CW_SOKOL}")
    target_compile_features(clockwork_r2d_fx_tests PRIVATE c_std_11)
    if(NOT MSVC)
        target_link_libraries(clockwork_r2d_fx_tests PRIVATE m)
    endif()
    add_test(NAME clockwork_r2d_fx_tests COMMAND clockwork_r2d_fx_tests)
    set_tests_properties(clockwork_r2d_fx_tests PROPERTIES PASS_REGULAR_EXPRESSION "OK: [0-9]+ particle/effect checks passed")
    # Post-process config layer: tunables, parameter table, order parsing, shake/motion math.
    add_executable(clockwork_r2d_post_tests tests/test_r2d_post.c render2d/r2d_post_params.c render2d/r2d_tunables.c render2d/r2d_shadow.c render2d/r2d_quality.c)
    target_include_directories(clockwork_r2d_post_tests PRIVATE render2d)
    target_include_directories(clockwork_r2d_post_tests SYSTEM PRIVATE "${CW_SOKOL}")
    target_compile_features(clockwork_r2d_post_tests PRIVATE c_std_11)
    if(NOT MSVC)
        target_link_libraries(clockwork_r2d_post_tests PRIVATE m)
    endif()
    add_test(NAME clockwork_r2d_post_tests COMMAND clockwork_r2d_post_tests)
    set_tests_properties(clockwork_r2d_post_tests PROPERTIES PASS_REGULAR_EXPRESSION "OK: [0-9]+ post-process config checks passed")
    if(CW_BUILD_EXAMPLES)
        add_test(NAME clockwork_movement COMMAND clockwork_movement)
        set_tests_properties(clockwork_movement PROPERTIES PASS_REGULAR_EXPRESSION "After 30 ticks: \\(3.00, 1.00\\)")
        if(TARGET clockwork_save_load)
            add_test(NAME clockwork_save_load COMMAND clockwork_save_load --headless ${CMAKE_CURRENT_BINARY_DIR}/clockwork_save_load_test.cwsn)
            set_tests_properties(clockwork_save_load PROPERTIES PASS_REGULAR_EXPRESSION "OK: save/load verified")
        endif()
        if(TARGET clockwork_combat)
            add_test(NAME clockwork_combat COMMAND clockwork_combat --headless ${CMAKE_CURRENT_BINARY_DIR}/clockwork_combat_test.cwsn)
            set_tests_properties(clockwork_combat PROPERTIES PASS_REGULAR_EXPRESSION "OK: combat demo verified")
        endif()
        if(UNIX)
            add_test(NAME clockwork_visualizer COMMAND clockwork_visualizer --frames 300)
            set_tests_properties(clockwork_visualizer PROPERTIES PASS_REGULAR_EXPRESSION "OK: [0-9]+ checks, 0 violations")
        endif()
    endif()
endif()

include(GNUInstallDirs)
install(TARGETS clockwork EXPORT clockworkTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/clockwork.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT clockworkTargets NAMESPACE clockwork:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/clockwork)
