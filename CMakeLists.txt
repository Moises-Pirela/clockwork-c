cmake_minimum_required(VERSION 3.16)
project(clockwork VERSION 1.0.0 LANGUAGES C)

option(CW_BUILD_TESTS "Build Clockwork regression tests" ON)
option(CW_BUILD_EXAMPLES "Build the movement example" ON)
option(CW_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

add_library(clockwork STATIC src/clockwork.c src/cw_sim.c src/cw_commands.c)
add_library(clockwork::clockwork ALIAS clockwork)
target_include_directories(clockwork PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
set_target_properties(clockwork PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_compile_features(clockwork PUBLIC c_std_11)
if(MSVC)
    target_compile_options(clockwork PRIVATE /W4 /WX)
else()
    target_compile_options(clockwork PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(CW_SANITIZE)
    if(MSVC)
        target_compile_options(clockwork PUBLIC /fsanitize=address)
    else()
        target_compile_options(clockwork PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(clockwork PUBLIC -fsanitize=address,undefined)
    endif()
endif()

if(CW_BUILD_EXAMPLES)
    add_executable(clockwork_movement examples/movement.c)
    target_link_libraries(clockwork_movement PRIVATE clockwork)
    if(UNIX)
        # Terminal visualizer (POSIX termios); run it headless with --frames N.
        add_executable(clockwork_visualizer examples/visualizer.c)
        target_link_libraries(clockwork_visualizer PRIVATE clockwork m)
    endif()
endif()
if(CW_BUILD_TESTS)
    enable_testing()
    add_executable(clockwork_tests tests/test_clockwork.c)
    target_link_libraries(clockwork_tests PRIVATE clockwork)
    add_test(NAME clockwork_tests COMMAND clockwork_tests)
    if(CW_BUILD_EXAMPLES)
        add_test(NAME clockwork_movement COMMAND clockwork_movement)
        set_tests_properties(clockwork_movement PROPERTIES PASS_REGULAR_EXPRESSION "After 30 ticks: \\(3.00, 1.00\\)")
        if(UNIX)
            add_test(NAME clockwork_visualizer COMMAND clockwork_visualizer --frames 300)
            set_tests_properties(clockwork_visualizer PROPERTIES PASS_REGULAR_EXPRESSION "OK: [0-9]+ checks, 0 violations")
        endif()
    endif()
endif()

include(GNUInstallDirs)
install(TARGETS clockwork EXPORT clockworkTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/clockwork.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT clockworkTargets NAMESPACE clockwork:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/clockwork)
